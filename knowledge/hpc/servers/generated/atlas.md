```bash
#!/bin/bash
# gaussian-09 (gaussian 09) on atlas:vanilla; generated by `rag-drg servers submit` from servers.yaml
# HTCondor job script, run by submit.sub (16 cores, 32 GB requested there);
# submit with: chmod u+x job.sh && condor_submit submit.sub
# expected run time 24:00:00; HTCondor has no walltime request, and the pool holds jobs running
# longer than 72:00:00 (vanilla in servers.yaml)

WORKDIR="${_CONDOR_JOB_IWD:-$PWD}"
JOBID="${CONDOR_JOBID:-$$}"
cd "$WORKDIR"

# --- gaussian-09: absolute paths, no environment modules ---

source $g09root/g09/bsd/g09.profile
GAUSSIAN=/Local/ce_dana/g09/g09

# --- per-job scratch ---
SCRATCH="/storage/ce_dana/$USER/scratch/$JOBID"

# on exit, also after a walltime kill or scancel/qdel/condor_rm (SIGTERM): copy back and clean up
cleanup() {
    cd "$WORKDIR" || true
    rm -rf "$SCRATCH"
}
trap cleanup EXIT
trap 'exit 143' TERM INT

export GAUSS_SCRDIR="$SCRATCH"
cd "$WORKDIR"
"$GAUSSIAN" < "job.gjf" > "job.log"
```
